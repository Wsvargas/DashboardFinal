# FORMATTERS COMUNES
# ──────────────────────────────────────────────────────────────

def fmt_manager(n, prefix="", suffix=""):
    """Formatea números grandes con unidades (M, mil, etc)"""
    if pd.isna(n):
        return "—"
    x = float(n)
    sign = "-" if x < 0 else ""
    x = abs(x)
    if x >= 1_000_000:
        txt = f"{x / 1_000_000:.1f} M"
    elif x >= 1_000:
        txt = f"{x / 1_000:.1f} mil"
    else:
        txt = f"{x:,.0f}"
    txt = txt.replace(".0 M", " M").replace(".0 mil", " mil")
    return f"{sign}{prefix}{txt}{suffix}"


def fmt_signed_short(n, prefix="", suffix=""):
    """Formatea números con signo (+ o -)"""
    if pd.isna(n):
        return "—"
    x = float(n)
    base = fmt_manager(abs(x), prefix=prefix, suffix=suffix)
    if x > 0:
        return f"+{base}"
    elif x < 0:
        return f"-{base}"
    return base

def render_kpis_globales(SF, context_label=None):
    """Renderiza los 6 KPIs globales superiores, filtrados por la selección activa."""
    kg_total = pd.to_numeric(SF["KgLive"], errors="coerce").sum()
    costo_total = pd.to_numeric(SF["CostoAcum"], errors="coerce").sum()
    cpkg = costo_total / (kg_total if pd.notna(kg_total) and kg_total > 0 else np.nan)

    if "Granja" in SF.columns:
        granjas_total = SF["Granja"].nunique()
    elif "GranjaID" in SF.columns:
        granjas_total = SF["GranjaID"].nunique()
    else:
        granjas_total = SF["NombreGranja"].nunique()

    if context_label:
        md(f'''
<div style="display:flex;align-items:center;gap:8px;margin-bottom:6px;">
  <span style="font-size:.7rem;color:{MUTED};font-weight:500;text-transform:uppercase;
               letter-spacing:.4px;">KPIs filtrando por</span>
  <span style="font-size:.75rem;font-weight:800;color:{RED};
               background:rgba(218,41,28,.08);border:1px solid rgba(218,41,28,.25);
               border-radius:4px;padding:1px 8px;">{context_label}</span>
  <span style="font-size:.66rem;color:{MUTED};opacity:.7;">
    · clic en área vacía del gráfico para limpiar
  </span>
</div>''')

    k1, k2, k3, k4, k5, k6 = st.columns(6)
    for col_, val_, lbl_, acc in [
        (k1, f"{SF['LoteCompleto'].nunique():,}", "Lotes activos", True),
        (k2, f"{granjas_total:,}", "Granjas totales", True),
        (k3, fmt_manager(pd.to_numeric(SF["AvesVivas"], errors="coerce").sum()), "Aves vivas", True),
        (k4, fmt_manager(kg_total, suffix=" kg"), "Kg live", True),
        (k5, fmt_manager(costo_total, prefix="$"), "Costo total", True),
        (k6, fmt_num(cpkg, 3, prefix="$", suffix="/kg"), "Costo medio/kg", False),
    ]:
        with col_:
            md(f'<div class="kpi-chip {"accent" if acc else ""}"><div class="kv">{val_}</div><div class="kl">{lbl_}</div></div>')


def render_kpi_sobrecosto(SF):
    """
    Banda ejecutiva: sobrecosto acumulado vs ideal comparable y
    ahorro potencial si los lotes con desvío operaran en su ideal.
    """
    if "GapCostoComp" not in SF.columns:
        return

    gap = pd.to_numeric(SF["GapCostoComp"], errors="coerce")
    gap_pos = gap[gap > 0]
    if gap_pos.empty:
        md(f'''
<div style="display:flex;align-items:center;gap:8px;margin:2px 0 2px 0;
            background:rgba(22,163,74,.07);border:1px solid rgba(22,163,74,.3);
            border-radius:8px;padding:4px 12px;font-size:.78rem;font-weight:700;color:{GREEN};">
  ✅ Ningún lote del filtro actual opera con sobrecosto vs su ideal comparable
</div>''')
        return

    total_sobrecosto = float(gap_pos.sum())
    n_lotes = int(gap_pos.count())

    md(f'''
<div style="display:flex;align-items:center;gap:10px;flex-wrap:wrap;margin:2px 0 2px 0;
            background:rgba(218,41,28,.06);border:1px solid rgba(218,41,28,.25);
            border-radius:8px;padding:4px 12px;font-size:.78rem;color:{TEXT};">
  💰 <b>Sobrecosto vs ideal:</b>
  <span style="font-weight:900;color:{RED};font-size:.88rem;">{fmt_manager(total_sobrecosto, prefix="$")}</span>
  en <b>{n_lotes}</b> lote(s)
  <span style="color:{MUTED};font-weight:400;">· ahorro potencial si operaran en su ideal</span>
</div>''')
